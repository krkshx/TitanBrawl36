cmake_minimum_required(VERSION 3.21)
project(TitanBrawl36 VERSION 36.0.0 LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Все выходные бинарники — в <build>/linux одной папкой.
set(TITAN_OUTDIR ${CMAKE_BINARY_DIR}/linux)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${TITAN_OUTDIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${TITAN_OUTDIR})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${TITAN_OUTDIR})

# Крипта: по умолчанию встроенный фолбэк (X25519+BLAKE2b, SodiumFallback.cpp).
# TITAN_WITH_SODIUM=ON — системный libsodium (нужен dev-пакет).
option(TITAN_WITH_SODIUM "Use system libsodium for crypto (needs dev package)" OFF)
option(TITAN_BUILD_LIBG "Build libg engine shared library" ON)

# libg — весь движок одной shared-библиотекой (unity TU: src/libg.cpp).
if(TITAN_BUILD_LIBG)
  add_library(g SHARED src/libg.cpp)
  set_target_properties(g PROPERTIES OUTPUT_NAME "g")
  target_include_directories(g PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  # Весь движок — header-style классы с методами в теле класса (implicit inline):
  # без этого флага линкер выбрасывает неиспользованные inline-методы и .so
  # получается пустой. Флаг заставляет эмитить их все в библиотеку.
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(g PRIVATE -fkeep-inline-functions)
  endif()
  if(UNIX)
    target_link_options(g PRIVATE "LINKER:--no-undefined")
  endif()
endif()

add_executable(titan_pc src/app/Main.cpp)
target_include_directories(titan_pc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
# На ELF дубли символов exe/.so легальны (свои перекрывают), exe реально
# зависит от libg.so. На Windows линковку пропускаем (LNK2005), exe автономен.
if(TITAN_BUILD_LIBG AND NOT WIN32)
  target_link_libraries(titan_pc PRIVATE g)
endif()

# exe и libg.so лежат рядом, ищут друг друга через $ORIGIN.
if(TITAN_BUILD_LIBG)
  set_target_properties(g PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
set_target_properties(titan_pc PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")

# Общие сторонние зависимости для целей.
function(titan_link_thirdparty tgt)
  if(TITAN_WITH_SODIUM)
    find_package(PkgConfig QUIET)
    if(PKG_CONFIG_FOUND)
      pkg_check_modules(SODIUM QUIET libsodium)
    endif()
    if(SODIUM_FOUND)
      target_include_directories(${tgt} PRIVATE ${SODIUM_INCLUDE_DIRS})
      target_link_libraries(${tgt} PRIVATE ${SODIUM_LIBRARIES})
    else()
      find_library(SODIUM_LIB sodium)
      if(SODIUM_LIB)
        target_link_libraries(${tgt} PRIVATE ${SODIUM_LIB})
      endif()
    endif()
    if(SODIUM_FOUND OR SODIUM_LIB)
      target_compile_definitions(${tgt} PRIVATE TITAN_WITH_SODIUM)
    else()
      message(WARNING "TITAN_WITH_SODIUM=ON but libsodium not found - using built-in fallback")
    endif()
  endif()

  find_package(Freetype QUIET)
  if(NOT Freetype_FOUND AND NOT FREETYPE_FOUND)
    # Вендорный FreeType: текст игры рендерится из assets/font/*.otf (Pusia-Bold),
    # а системного freetype на Windows обычно нет — собираем свой, статикой.
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/freetype/CMakeLists.txt)
      set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
      set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
      set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
      set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
      set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
      set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
      add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/third_party/freetype
                       ${CMAKE_BINARY_DIR}/freetype)
      set(TITAN_BUNDLED_FREETYPE ON)
    endif()
  endif()
  if(Freetype_FOUND OR FREETYPE_FOUND OR TITAN_BUNDLED_FREETYPE OR TARGET freetype)
    if(TARGET Freetype::Freetype)
      target_link_libraries(${tgt} PRIVATE Freetype::Freetype)
    elseif(TARGET freetype)
      target_link_libraries(${tgt} PRIVATE freetype)
    else()
      target_include_directories(${tgt} PRIVATE ${FREETYPE_INCLUDE_DIRS})
      target_link_libraries(${tgt} PRIVATE ${FREETYPE_LIBRARIES})
    endif()
    target_compile_definitions(${tgt} PRIVATE TITAN_HAS_FREETYPE)
  endif()

  if(UNIX AND NOT APPLE)
    find_package(X11 REQUIRED)
    target_include_directories(${tgt} PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(${tgt} PRIVATE ${X11_LIBRARIES})
    find_package(PkgConfig QUIET)
    if(PKG_CONFIG_FOUND)
      pkg_check_modules(VORBISFILE QUIET vorbisfile)
      pkg_check_modules(PULSE_SIMPLE QUIET libpulse-simple)
      pkg_check_modules(XFT QUIET xft)
    endif()
    pkg_check_modules(LIBLZMA QUIET liblzma)
    if(LIBLZMA_FOUND)
      target_include_directories(${tgt} PRIVATE ${LIBLZMA_INCLUDE_DIRS})
      target_link_libraries(${tgt} PRIVATE ${LIBLZMA_LIBRARIES})
      target_compile_definitions(${tgt} PRIVATE TITAN_HAS_LZMA)
    else()
      find_package(LibLZMA QUIET)
      if(LIBLZMA_FOUND)
        target_link_libraries(${tgt} PRIVATE LibLZMA::LibLZMA)
        target_compile_definitions(${tgt} PRIVATE TITAN_HAS_LZMA)
      endif()
    endif()
    if(VORBISFILE_FOUND AND PULSE_SIMPLE_FOUND)
      target_include_directories(${tgt} PRIVATE ${VORBISFILE_INCLUDE_DIRS} ${PULSE_SIMPLE_INCLUDE_DIRS})
      target_link_libraries(${tgt} PRIVATE ${VORBISFILE_LIBRARIES} ${PULSE_SIMPLE_LIBRARIES})
      target_compile_definitions(${tgt} PRIVATE TITAN_HAS_AUDIO)
    endif()
    if(XFT_FOUND)
      target_include_directories(${tgt} PRIVATE ${XFT_INCLUDE_DIRS})
      target_link_libraries(${tgt} PRIVATE ${XFT_LIBRARIES})
    else()
      find_library(XFT_LIB Xft)
      if(XFT_LIB)
        target_link_libraries(${tgt} PRIVATE ${XFT_LIB})
      endif()
    endif()
  endif()

  if(WIN32)
    target_link_libraries(${tgt} PRIVATE winmm ws2_32)
  endif()

  if(MSVC)
    target_compile_options(${tgt} PRIVATE /W4)
  else()
    target_compile_options(${tgt} PRIVATE -Wall -Wextra)
  endif()
endfunction()

if(TARGET g)
  titan_link_thirdparty(g)
endif()
titan_link_thirdparty(titan_pc)
