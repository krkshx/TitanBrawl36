cmake_minimum_required(VERSION 3.21)
project(TitanBrawl36 VERSION 36.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(titan_pc src/app/Main.cpp)
target_include_directories(titan_pc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
find_package(Freetype QUIET)
if(Freetype_FOUND OR FREETYPE_FOUND)
  if(TARGET Freetype::Freetype)
    target_link_libraries(titan_pc PRIVATE Freetype::Freetype)
  else()
    target_include_directories(titan_pc PRIVATE ${FREETYPE_INCLUDE_DIRS})
    target_link_libraries(titan_pc PRIVATE ${FREETYPE_LIBRARIES})
  endif()
  target_compile_definitions(titan_pc PRIVATE TITAN_HAS_FREETYPE)
endif()
if(MSVC)
  target_compile_options(titan_pc PRIVATE /W4)
else()
  target_compile_options(titan_pc PRIVATE -Wall -Wextra)
endif()
if(UNIX AND NOT APPLE)
  find_package(X11 REQUIRED)
  target_include_directories(titan_pc PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(titan_pc PRIVATE ${X11_LIBRARIES})
  find_package(PkgConfig QUIET)
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(VORBISFILE QUIET vorbisfile)
    pkg_check_modules(PULSE_SIMPLE QUIET libpulse-simple)
    pkg_check_modules(XFT QUIET xft)
  endif()
  if(VORBISFILE_FOUND AND PULSE_SIMPLE_FOUND)
    target_include_directories(titan_pc PRIVATE ${VORBISFILE_INCLUDE_DIRS} ${PULSE_SIMPLE_INCLUDE_DIRS})
    target_link_libraries(titan_pc PRIVATE ${VORBISFILE_LIBRARIES} ${PULSE_SIMPLE_LIBRARIES})
    target_compile_definitions(titan_pc PRIVATE TITAN_HAS_AUDIO)
  endif()
  if(XFT_FOUND)
    target_include_directories(titan_pc PRIVATE ${XFT_INCLUDE_DIRS})
    target_link_libraries(titan_pc PRIVATE ${XFT_LIBRARIES})
  else()
    find_library(XFT_LIB Xft)
    if(XFT_LIB)
      target_link_libraries(titan_pc PRIVATE ${XFT_LIB})
    endif()
  endif()
endif()
if(WIN32)
  target_link_libraries(titan_pc PRIVATE winmm)
endif()
